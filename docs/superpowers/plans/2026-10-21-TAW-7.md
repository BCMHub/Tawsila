import { describe, expect, it } from '@jest/globals';
import { createSecureStorage } from '../../features/auth/secureStorage';

function fakeStore() {
  const m = new Map<string, string>();
  const s = {
    m, writes: 0, failAt: 0,
    getItemAsync: async (k: string) => m.get(k) ?? null,
    setItemAsync: async (k: string, v: string) => { s.writes += 1; if (s.failAt && s.writes === s.failAt) throw new Error('interrupted'); m.set(k, v); },
    deleteItemAsync: async (k: string) => { m.delete(k); },
  };
  return s;
}
const OLD = 'o'.repeat(450);
const NEW = 'n'.repeat(520);
const opts = { chunkSize: 100 };

async function cleanKeyCount(value: string) {
  const store = fakeStore();
  await createSecureStorage(store, opts).setItem('k', value);
  return store.m.size;
}

describe('secure session storage', () => {
  it('round-trips a multi-chunk value and never writes an entry larger than the chunk size (manifest excluded from the bound only if it is smaller)', async () => {
    const store = fakeStore();
    const s = createSecureStorage(store, opts);
    await s.setItem('k', NEW);
    expect(await s.getItem('k')).toBe(NEW);
    for (const v of store.m.values()) expect(v.length).toBeLessThanOrEqual(100);
  });

  it('returns null for an unknown key and rejects non-ASCII values', async () => {
    const s = createSecureStorage(fakeStore(), opts);
    expect(await s.getItem('missing')).toBeNull();
    await expect(s.setItem('k', 'é')).rejects.toThrow();
  });

  it('an interrupted replacement at every write boundary leaves the previous session readable and recoverable without orphans', async () => {
    const probe = fakeStore();
    const ps = createSecureStorage(probe, opts);
    await ps.setItem('k', OLD);
    const before = probe.writes;
    await ps.setItem('k', NEW);
    const writesPerReplacement = probe.writes - before;
    expect(writesPerReplacement).toBeGreaterThan(1);

    for (let n = 1; n <= writesPerReplacement; n++) {
      const store = fakeStore();
      const s = createSecureStorage(store, opts);
      await s.setItem('k', OLD);
      store.failAt = store.writes + n;
      await expect(s.setItem('k', NEW)).rejects.toThrow();
      expect(await s.getItem('k')).toBe(OLD); // never a mix, never lost
      store.failAt = 0;
      await s.setItem('k', NEW);
      expect(await s.getItem('k')).toBe(NEW);
      expect(store.m.size).toBe(await cleanKeyCount(NEW)); // orphaned chunks were pruned
    }
  });

  it('a first write interrupted at any boundary reads as no session and leaves a clean retry', async () => {
    const probe = fakeStore();
    await createSecureStorage(probe, opts).setItem('k', NEW);
    const total = probe.writes;
    for (let n = 1; n <= total; n++) {
      const store = fakeStore();
      const s = createSecureStorage(store, opts);
      store.failAt = n;
      await expect(s.setItem('k', NEW)).rejects.toThrow();
      expect(await s.getItem('k')).toBeNull();
      store.failAt = 0;
      await s.setItem('k', NEW);
      expect(await s.getItem('k')).toBe(NEW);
      expect(store.m.size).toBe(await cleanKeyCount(NEW));
    }
  });

  it('a missing or corrupted entry never yields a partial or mixed value, and the adapter recovers', async () => {
    const entries = (await (async () => { const st = fakeStore(); await createSecureStorage(st, opts).setItem('k', NEW); return st.m.size; })());
    expect(entries).toBeGreaterThan(1);
    for (let i = 0; i < entries; i++) {
      for (const mode of ['delete', 'garbage'] as const) {
        const store = fakeStore();
        const s = createSecureStorage(store, opts);
        await s.setItem('k', NEW);
        const damaged = [...store.m.keys()][i]; // chosen from the very store being damaged
        expect(damaged).toBeDefined();
        if (mode === 'delete') store.m.delete(damaged); else store.m.set(damaged, 'garbage');
        const r = await s.getItem('k');
        expect(r === null || r === NEW).toBe(true);
        await s.setItem('k', OLD);
        expect(await s.getItem('k')).toBe(OLD);
        expect(store.m.size).toBe(await cleanKeyCount(OLD));
      }
    }
  });

  it('removal leaves nothing behind, including after earlier replacements', async () => {
    const store = fakeStore();
    const s = createSecureStorage(store, opts);
    await s.setItem('k', OLD);
    await s.setItem('k', NEW);
    await s.removeItem('k');
    expect(store.m.size).toBe(0);
    expect(await s.getItem('k')).toBeNull();
  });

  it('concurrent writes are serialized: the result is one complete value and no orphans remain', async () => {
    const store = fakeStore();
    const s = createSecureStorage(store, opts);
    await Promise.all([s.setItem('k', OLD), s.setItem('k', NEW)]);
    const r = await s.getItem('k');
    expect(r === OLD || r === NEW).toBe(true);
    expect(store.m.size).toBe(await cleanKeyCount(r as string));
  });
});

Specifically: state derivation (no session: `signed_out`; attempt running: `authenticating`; session and own-row read pending: `checking`; row missing or read failed: `account_provision_error` with `canRetry`; row with `profile_completed_at` null: `needs_profile`; set: `ready_to_browse`); the six session invariants of spec section 4; the Apple name update is optional; the nonce raw value goes to `signInWithIdToken`; a fresh installation signs out and clears the stored session while a normal update keeps it, and a failing guard fails toward sign-in; no tabs on a provider event alone; nothing happens after unmount; the SDK session is authoritative: **on an auth `SIGNED_OUT` event the controller re-reads the SDK session with `getSession`; a session that still exists means the event is stale and is ignored; no session means sign-out (once); a failing read (network error or HTTP 5xx) clears nothing and shows a notice; an `AuthRefreshDiscardedError` is a successful no-op (nothing cleared, no notice)**; a late result of an operation that already timed out or was unmounted changes nothing. **Abandonment boundary:** the app starts no exchange, read or write once it is unmounted or the attempt has expired; a token exchange that the SDK has already been handed cannot be cancelled, so its completion is the SDK's own change (the user did complete the provider flow and the SDK keeps that session), not an app-side modification, and the app performs no further action for it (no account read, no name update, no sign-out, no state or navigation change); the next launch derives the state from the SDK session as usual. ### Task 6: Provider wrappers, static guards (can start now; real flows blocked)

**Files:** `apps/mobile/src/features/auth/providers.ts`, `apps/mobile/src/__tests__/auth/noOtp.test.ts`.

**Interface (fixed):** `providers.ts` exports `createProviderHandlers(platform: 'ios' | 'android', deps?: ProviderDeps): ProviderHandlers` where
```ts
export type AppleDeps = { isAvailableAsync(): Promise<boolean>; signInAsync(o: { nonce: string }): Promise<{ identityToken: string | null; fullName?: { givenName?: string | null; familyName?: string | null } | null }> };
export type GoogleDeps = { signIn(o: { nonce: string }): Promise<{ idToken: string | null }> }; // the adapter over the validated library throws { code: 'cancelled' } on user cancellation
export type ProviderDeps = { apple?: AppleDeps; google?: GoogleDeps; randomNonce?: () => string; sha256Hex?: (s: string) => Promise<string> };
```
Google is offered on both platforms, Apple only on iOS (and only when `isAvailableAsync()` is true). Each call generates a fresh random raw nonce, gives its SHA-256 hex to the provider and returns `{ idToken, nonce: raw }`; it maps provider cancellation to `{ code: 'cancelled' }` (Apple's `ERR_REQUEST_CANCELED`), denial to `{ code: 'denied' }` and unavailability to `{ code: 'unavailable' }`; an empty or null token rejects; it never logs tokens. Native modules are imported lazily by the default `deps`, so unit tests need no native build. **The default Google deps are written after the D1 spike (Task 9) with the validated library; the Apple nonce path follows the device verification.**

```ts
import { beforeAll, describe, expect, it } from '@jest/globals';
import { createClient } from '@supabase/supabase-js';

const env = { url: process.env.SUPABASE_URL, anon: process.env.SUPABASE_ANON_KEY,
  token1: process.env.TEST_GOOGLE_ID_TOKEN, nonce1: process.env.TEST_GOOGLE_RAW_NONCE, token2: process.env.TEST_GOOGLE_ID_TOKEN_2, nonce2: process.env.TEST_GOOGLE_RAW_NONCE_2 };
beforeAll(() => {
  for (const [k, v] of Object.entries(env)) if (!v) throw new Error(`Set ${k} (see supabase/README.md); real test identities only`);
});
const mk = () => createClient(env.url!, env.anon!, { auth: { persistSession: false, autoRefreshToken: false } });

describe('Google ID-token sign-in against the test project (test identity, tokens from the spike)', () => {
  it('rejects a token presented with a different nonce and creates no session', async () => {
    const c = mk();
    const r = await c.auth.signInWithIdToken({ provider: 'google', token: env.token1!, nonce: `${env.nonce1}-wrong` });
    expect(r.error).not.toBeNull();
    expect((await c.auth.getSession()).data.session).toBeNull();
  });
  it('accepts the matching nonce, creates exactly one account row, and a returning sign-in reaches the same user without a second row', async () => {
    const c = mk();
    const first = await c.auth.signInWithIdToken({ provider: 'google', token: env.token1!, nonce: env.nonce1! });
    expect(first.error).toBeNull();
    const id = first.data.user!.id;
    const rows1 = await c.from('accounts').select('user_id').eq('user_id', id);
    expect(rows1.data).toHaveLength(1);
    await c.auth.signOut();
    const second = await c.auth.signInWithIdToken({ provider: 'google', token: env.token2!, nonce: env.nonce2! });
    expect(second.error).toBeNull();
    expect(second.data.user!.id).toBe(id);
    const rows2 = await c.from('accounts').select('user_id').eq('user_id', id);
    expect(rows2.data).toHaveLength(1);
  });
});
```

- [ ] **Step 2c: Concurrent first sign-ins (database level, needs D7 or D4).** Script `supabase/tests/concurrency/first-signin.sh` (complete): starts 20 parallel transactions each inserting a distinct new `auth.users` row and verifies exactly 20 `accounts` rows were created, one per user, then cleans up only its own users:

```bash
#!/usr/bin/env bash
set -euo pipefail
DB="${SUPABASE_DB_CONTAINER:-supabase_db_tawsila}"
psqlc() { docker exec -i "$DB" psql -U postgres -d postgres -qtAX "$@"; }
ids=$(for i in $(seq 1 20); do uuidgen | tr 'A-Z' 'a-z'; done)
trap 'for id in $ids; do psqlc -c "delete from auth.users where id = '"'"'$id'"'"'" >/dev/null; done' EXIT
echo "$ids" | xargs -P 20 -I{} docker exec -i "$DB" psql -U postgres -d postgres -qtAX -c "insert into auth.users (id, aud, role) values ('{}', 'authenticated', 'authenticated');" >/dev/null
count=$(psqlc -c "select count(*) from public.accounts where user_id in ($(echo "$ids" | sed "s/.*/'&'/" | paste -sd, -))")
users=$(psqlc -c "select count(*) from auth.users where id in ($(echo "$ids" | sed "s/.*/'&'/" | paste -sd, -))")
[ "$count" = "20" ] && [ "$users" = "20" ] && echo "PASS 20 users, 20 account rows" || { echo "FAIL users=$users accounts=$count"; exit 1; }
```
