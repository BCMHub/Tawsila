# TAW-7 Account sign-up implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new user registers with a mobile number (default +20), gets an anonymous Supabase session and an unverified `accounts` row, and reaches the tabs, with no OTP and no SMS.

**Architecture:** `@supabase/supabase-js` in the Expo app; a Supabase project defined as code in `supabase/` (migrations, config, pgTAP tests, a concurrency script). One `security definer` function, `register_phone`, is the only write path to `public.accounts`; a counter table gives an atomic server-side rate limit. The app derives a three-state registration model (signed-out, needs-phone, registered) in an `AuthProvider` and routes with `Stack.Protected`.

**Tech stack:** Expo SDK 58, React Native 0.88, TypeScript strict, Expo Router, jest-expo, `@supabase/supabase-js`, `react-native-url-polyfill`, `expo-secure-store`, Supabase CLI with a local Docker stack, pgTAP.

### Task 3: Migration (make the database tests pass)

**Files:** `supabase/migrations/20261021000000_accounts.sql`.

- [ ] **Step 1: Write the migration.**

```sql
-- TAW-7: accounts with an unverified phone, and the only write path to it.
create table public.accounts (
  user_id uuid primary key references auth.users (id) on delete cascade,
  phone_e164 text not null check (phone_e164 ~ '^\+[1-9][0-9]{7,14}$'),
  phone_verified boolean not null default false,
  created_at timestamptz not null default now()
);

-- Uniqueness only among verified numbers (TAW-5 sets phone_verified through trusted code).
create unique index accounts_verified_phone_key on public.accounts (phone_e164) where phone_verified;

alter table public.accounts enable row level security;
revoke all on public.accounts from anon, authenticated;
grant select on public.accounts to authenticated;
create policy accounts_select_own on public.accounts
  for select to authenticated using (user_id = (select auth.uid()));

create table public.registration_limits (
  key text not null,
  window_start timestamptz not null,
  attempts integer not null,
  primary key (key, window_start)
);
alter table public.registration_limits enable row level security;
revoke all on public.registration_limits from anon, authenticated;

create function public.register_phone(p_phone text)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_uid uuid := auth.uid();
  v_ip text;
  v_phone text;
  v_user_attempts integer;
  v_ip_attempts integer := 0;
  v_stored text;
begin
  if v_uid is null then
    raise exception 'not authenticated' using errcode = '28000';
  end if;

  -- Atomic counters: the conflict update takes a row lock, so concurrent callers are serialized per key.
  insert into public.registration_limits as l (key, window_start, attempts)
  values ('u:' || v_uid::text, date_trunc('hour', now()), 1)
  on conflict (key, window_start) do update set attempts = l.attempts + 1
  returning l.attempts into v_user_attempts;

  v_ip := nullif(trim(split_part(coalesce((nullif(current_setting('request.headers', true), '')::json ->> 'x-forwarded-for'), ''), ',', 1)), '');
  if v_ip is not null then
    insert into public.registration_limits as l (key, window_start, attempts)
    values ('ip:' || v_ip, date_trunc('hour', now()), 1)
    on conflict (key, window_start) do update set attempts = l.attempts + 1
    returning l.attempts into v_ip_attempts;
  end if;

  delete from public.registration_limits where window_start < now() - interval '24 hours';

  -- Return, never raise, so the counter increments commit.
  if v_user_attempts > 5 or v_ip_attempts > 20 then
    return jsonb_build_object('ok', false, 'reason', 'rate_limited');
  end if;

  v_phone := regexp_replace(coalesce(p_phone, ''), '[\s\-().]', '', 'g');
  if v_phone !~ '^\+[1-9][0-9]{7,14}$'
     or (v_phone ~ '^\+20' and v_phone !~ '^\+20(10|11|12|15)[0-9]{8}$') then
    return jsonb_build_object('ok', false, 'reason', 'invalid_number');
  end if;

  insert into public.accounts (user_id, phone_e164)
  values (v_uid, v_phone)
  on conflict (user_id) do update set phone_e164 = excluded.phone_e164
    where public.accounts.phone_verified = false;

  select a.phone_e164 into v_stored from public.accounts a where a.user_id = v_uid;

  if v_stored = v_phone then
    return jsonb_build_object('ok', true);
  end if;
  return jsonb_build_object('ok', false, 'reason', 'already_verified');
end;
$$;

revoke all on function public.register_phone(text) from public, anon;
grant execute on function public.register_phone(text) to authenticated;
```

Note: the `x-forwarded-for` header may be client-controlled behind some proxies, so the address limit is best effort; the per-user limit and the Auth `anonymous_users` limit are the primary controls (recorded in `supabase/README.md`, Task 12).

```sql
-- TAW-7 cleanup policy (spec section 8). Callable only by the owner; the scheduler is a recorded follow-up.
-- Activity signal: latest auth.sessions.refreshed_at. A user with no session timestamp is never deleted.
-- Never deleted: verified accounts, users that are no longer anonymous, users active within the window.
create function public.delete_stale_anonymous_users()
returns integer
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_deleted integer;
begin
  with signal as (
    select u.id as user_id, max(s.refreshed_at) as last_active
    from auth.users u left join auth.sessions s on s.user_id = u.id
    group by u.id
  ), d as (
    delete from auth.users u using signal g
    where g.user_id = u.id and u.is_anonymous and g.last_active is not null
      and not exists (select 1 from public.accounts a where a.user_id = u.id and a.phone_verified)
      and (
        (not exists (select 1 from public.accounts a where a.user_id = u.id)
          and u.created_at < now() - interval '24 hours' and g.last_active < now() - interval '24 hours')
        or
        (exists (select 1 from public.accounts a where a.user_id = u.id)
          and g.last_active < now() - interval '30 days')
      )
    returning 1
  )
  select count(*)::integer into v_deleted from d;
  return v_deleted;
end;
$$;
revoke all on function public.delete_stale_anonymous_users() from public, anon, authenticated;
```

```ts
export function normalizePhone(dial: string, national: string): PhoneResult {
```

```ts
import * as SecureStore from 'expo-secure-store';

type Store = Pick<typeof SecureStore, 'getItemAsync' | 'setItemAsync' | 'deleteItemAsync'>;

// Values are split so that no single secure-store entry exceeds the historical 2048-byte limit.
export function createSecureStorage(store: Store, chunkSize = 1800) {
  const countKey = (k: string) => `${k}.n`;
  const partKey = (k: string, i: number) => `${k}.${i}`;

  async function removeItem(key: string): Promise<void> {
    const n = Number(await store.getItemAsync(countKey(key)));
    await store.deleteItemAsync(countKey(key));
    for (let i = 0; i < (Number.isFinite(n) ? n : 0); i++) await store.deleteItemAsync(partKey(key, i));
  }

  return {
    async getItem(key: string): Promise<string | null> {
      const raw = await store.getItemAsync(countKey(key));
      if (raw === null) return null;
      const n = Number(raw);
      let out = '';
      for (let i = 0; i < n; i++) {
        const part = await store.getItemAsync(partKey(key, i));
        if (part === null) return null;
        out += part;
      }
      return out;
    },
    async setItem(key: string, value: string): Promise<void> {
      if (!/^[\x00-\x7F]*$/.test(value)) throw new Error('secure storage: non-ASCII value');
      await removeItem(key);
      const n = Math.max(1, Math.ceil(value.length / chunkSize));
      for (let i = 0; i < n; i++) await store.setItemAsync(partKey(key, i), value.slice(i * chunkSize, (i + 1) * chunkSize));
      await store.setItemAsync(countKey(key), String(n)); // written last: a partial write is never read
    },
    removeItem,
  };
}

export const secureStorage = createSecureStorage(SecureStore);
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';

export type RegisterOutcome =
  | { status: 'registered' }
  | { status: 'invalid_number' | 'rate_limited' | 'already_verified' | 'offline' | 'failed' };

export type RegistrationState =
  | { kind: 'signed-out' }
  | { kind: 'needs-phone'; userId: string }
  | { kind: 'registered'; userId: string };

export type ConfirmResult = RegistrationState | { kind: 'unchanged'; reason: 'offline' | 'failed' };

export type MarkerStore = { get(): Promise<string | null>; set(userId: string): Promise<void>; clear(): Promise<void> };
type Client = Pick<SupabaseClient, 'auth' | 'rpc' | 'from'>;
type ErrLike = { name?: string; message?: string; code?: string } | null | undefined;

export function isNetworkError(e: ErrLike): boolean {
  return !!e && (e.name === 'AuthRetryableFetchError' || /network request failed|failed to fetch|timed out/i.test(e.message ?? ''));
}
function isAuthorizationError(e: ErrLike, status?: number): boolean {
  return status === 401 || (!!e && (e.code === 'PGRST301' || /jwt/i.test(e.message ?? '')));
}
function isTemporary(e: ErrLike, status?: number): boolean {
  return isNetworkError(e) || (typeof status === 'number' && status >= 500);
}
const reasonOf = (e: ErrLike): 'offline' | 'failed' => (isNetworkError(e) ? 'offline' : 'failed');

export async function registerPhone(client: Client, e164: string): Promise<RegisterOutcome> {
  const { data: current } = await client.auth.getSession();
  if (!current.session) {
    const { error } = await client.auth.signInAnonymously();
    if (error) return { status: isNetworkError(error) ? 'offline' : 'failed' };
  }
  const { data, error } = await client.rpc('register_phone', { p_phone: e164 });
  if (error) return { status: isNetworkError(error) ? 'offline' : 'failed' };
  if (data?.ok === true) return { status: 'registered' };
  const reason = data?.reason;
  if (reason === 'invalid_number' || reason === 'rate_limited' || reason === 'already_verified') return { status: reason };
  return { status: 'failed' };
}

// First render uses local state only: the marker bounds the wait, and temporary failures never clear anything.
export async function readInitialState(client: Client, marker: MarkerStore, timeoutMs = 3000): Promise<RegistrationState> {
  const stored = await marker.get();
  let timer: ReturnType<typeof setTimeout> | undefined;
  const result = await Promise.race([
    client.auth.getSession().then((r) => ({ kind: 'done' as const, r })),
    new Promise<{ kind: 'timeout' }>((resolve) => { timer = setTimeout(() => resolve({ kind: 'timeout' }), timeoutMs); }),
  ]);
  if (timer) clearTimeout(timer);
  if (result.kind === 'done' && result.r.data.session) {
    const id = result.r.data.session.user.id;
    return stored === id ? { kind: 'registered', userId: id } : { kind: 'needs-phone', userId: id };
  }
  const temporary = result.kind === 'timeout' || isTemporary(result.r.error);
  if (temporary && stored) return { kind: 'registered', userId: stored };
  return { kind: 'signed-out' };
}

async function readOwnRow(client: Client, userId: string) {
  return client.from('accounts').select('user_id').eq('user_id', userId).maybeSingle();
}

async function terminate(client: Client, marker: MarkerStore): Promise<ConfirmResult> {
  await client.auth.signOut({ scope: 'local' });
  await marker.clear();
  return { kind: 'signed-out' };
}

export async function confirmRegistration(client: Client, userId: string, marker: MarkerStore): Promise<ConfirmResult> {
  let res = await readOwnRow(client, userId);
  if (res.error && isAuthorizationError(res.error, res.status)) {
    const refreshed = await client.auth.refreshSession();
    if (refreshed.error) {
      if (isTemporary(refreshed.error)) return { kind: 'unchanged', reason: reasonOf(refreshed.error) };
      return terminate(client, marker);
    }
    res = await readOwnRow(client, userId);
    if (res.error && isAuthorizationError(res.error, res.status)) return terminate(client, marker);
  }
  if (res.error) return { kind: 'unchanged', reason: reasonOf(res.error) };
  if (res.data) {
    await marker.set(userId);
    return { kind: 'registered', userId };
  }
  await marker.clear();
  return { kind: 'needs-phone', userId };
}
```

```tsx
import { createContext, useContext, useEffect, useRef, useState, type ReactNode } from 'react';
import { marker as defaultMarker, supabase as defaultClient } from '../../lib/supabase';
import {
  confirmRegistration, isNetworkError, readInitialState, type MarkerStore, type RegistrationState,
} from './registrationService';

type Notice = 'offline' | 'failed' | null;
type Ctx = { state: RegistrationState | null; notice: Notice; setRegistered: (userId: string) => void };
type Client = typeof defaultClient;

const AuthContext = createContext<Ctx>({ state: null, notice: null, setRegistered: () => undefined });
export const useAuth = () => useContext(AuthContext);

export function AuthProvider({ children, client = defaultClient, markerStore = defaultMarker }: { children: ReactNode; client?: Client; markerStore?: MarkerStore }) {
  const [state, setState] = useState<RegistrationState | null>(null);
  const [notice, setNotice] = useState<Notice>(null);
  const generation = useRef(0); // bumped on every state change so stale confirmations are ignored
  const current = useRef<RegistrationState | null>(null);

  const apply = (next: RegistrationState) => { generation.current += 1; current.current = next; setState(next); };

  useEffect(() => {
    let alive = true;

    async function confirm(from: RegistrationState) {
      if (from.kind === 'signed-out') return;
      const g = generation.current;
      const result = await confirmRegistration(client, from.userId, markerStore);
      if (!alive || g !== generation.current) return; // stale: a newer transition already happened
      if (result.kind === 'unchanged') { setNotice(result.reason); return; }
      setNotice(null);
      apply(result);
    }

    async function terminal() {
      const refreshed = await client.auth.refreshSession();
      if (!alive) return;
      if (!refreshed.error) return;
      if (isNetworkError(refreshed.error)) { setNotice('offline'); return; }
      await client.auth.signOut({ scope: 'local' });
      await markerStore.clear();
      if (alive) { setNotice(null); apply({ kind: 'signed-out' }); }
    }

    (async () => {
      const initial = await readInitialState(client, markerStore);
      if (!alive) return;
      apply(initial);
      await confirm(initial);
    })();

    const { data } = client.auth.onAuthStateChange((event, session) => {
      if (event === 'SIGNED_IN' && session && current.current?.kind === 'signed-out') {
        apply({ kind: 'needs-phone', userId: session.user.id });
      }
      if (event === 'SIGNED_OUT') void terminal();
    });

    return () => { alive = false; data.subscription.unsubscribe(); };
  }, [client, markerStore]);

  const setRegistered = (userId: string) => {
    apply({ kind: 'registered', userId });
    setNotice(null);
    void markerStore.set(userId);
  };

  return <AuthContext.Provider value={{ state, notice, setRegistered }}>{children}</AuthContext.Provider>;
}
```

```tsx
import { SafeAreaView } from 'react-native-safe-area-context';
import { NavBar } from '../../components/NavBar';
import { PhoneField } from '../../components/PhoneField';
import { PrimaryButton } from '../../components/PrimaryButton';
import { useAuth } from '../../features/auth/AuthProvider';

import { registerPhone } from '../../features/auth/registrationService';
import { supabase } from '../../lib/supabase';
import { strings } from '../../strings';
import { typography, useTheme } from '../../theme';

const messages: Record<string, string> = {
  invalid_number: strings.invalidNumber, rate_limited: strings.rateLimited, already_verified: strings.alreadyVerified,
  offline: strings.registerOffline, failed: strings.registerFailed,
};

export default function PhoneScreen() {
  const t = useTheme();
  const { state, setRegistered } = useAuth();
  const [country, setCountry] = useState<Country>(defaultCountry);
  const [value, setValue] = useState('');
  const [error, setError] = useState<string | null>(null);
  const [pending, setPending] = useState(false);

  async function onContinue() {
    if (pending) return;
    const phone = normalizePhone(country.dial, value);
    if (!phone.ok) { setError(strings.invalidNumber); return; }
    setError(null);
    setPending(true);
    const outcome = await registerPhone(supabase, phone.e164); // the number is never logged
    setPending(false);
    if (outcome.status === 'registered') {
      const { data } = await supabase.auth.getSession();
      if (data.session) setRegistered(data.session.user.id);
      return;
    }
    setError(messages[outcome.status] ?? strings.registerFailed);
  }

  return (
    <SafeAreaView style={{ flex: 1, backgroundColor: t['bg/base'] }}>
      <NavBar title="" variant="inline" onBack={state?.kind === 'needs-phone' ? undefined : () => router.back()} />
      <ScrollView keyboardShouldPersistTaps="handled" contentContainerStyle={{ flexGrow: 1, padding: 24, gap: 24 }}>
        <View style={{ gap: 8 }}>
          <Text accessibilityRole="header" style={{ ...typography.title1, color: t['label/primary'] }}>{strings.phoneTitle}</Text>
          <Text style={{ ...typography.body, color: t['label/secondary'] }}>{strings.phoneSubtitle}</Text>
        </View>
        <PhoneField country={country} onCountryChange={setCountry} value={value} onChangeText={setValue} error={error} editable={!pending} />
        <PrimaryButton label={strings.continue} onPress={onContinue} loading={pending} />
      </ScrollView>
    </SafeAreaView>
  );
}
```
