# TAW-7 Account sign-up implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new user registers with a mobile number (default +20), gets an anonymous Supabase session and an unverified `accounts` row, and reaches the tabs, with no OTP and no SMS.

**Architecture:** `@supabase/supabase-js` in the Expo app; a Supabase project defined as code in `supabase/` (migrations, config, pgTAP tests). One `security definer` function, `register_phone`, is the only write path to `public.accounts`; a counter table gives an atomic server-side rate limit. The app routes by a three-state registration model (signed-out, needs-phone, registered) using `Stack.Protected`.

**Tech stack:** Expo SDK 58, React Native 0.88, TypeScript strict, Expo Router, jest-expo, `@supabase/supabase-js`, `react-native-url-polyfill`, `expo-secure-store`, Supabase CLI with a local Docker stack, pgTAP.

- [ ] **Step 1: Write the migration**

```sql
-- TAW-7: accounts with an unverified phone, and the only write path to it.
create table public.accounts (
  user_id uuid primary key references auth.users (id) on delete cascade,
  phone_e164 text not null check (phone_e164 ~ '^\+[1-9][0-9]{7,14}$'),
  phone_verified boolean not null default false,
  created_at timestamptz not null default now()
);

-- Uniqueness only among verified numbers (TAW-5 sets phone_verified through trusted code).
create unique index accounts_verified_phone_key on public.accounts (phone_e164) where phone_verified;

alter table public.accounts enable row level security;
revoke all on public.accounts from anon, authenticated;
grant select on public.accounts to authenticated;
create policy accounts_select_own on public.accounts
  for select to authenticated using (user_id = (select auth.uid()));

create table public.registration_limits (
  key text not null,
  window_start timestamptz not null,
  attempts integer not null,
  primary key (key, window_start)
);
alter table public.registration_limits enable row level security;
revoke all on public.registration_limits from anon, authenticated;

create function public.register_phone(p_phone text)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_uid uuid := auth.uid();
  v_ip text;
  v_phone text;
  v_user_attempts integer;
  v_ip_attempts integer := 0;
  v_stored text;
  v_verified boolean;
begin
  if v_uid is null then
    raise exception 'not authenticated' using errcode = '28000';
  end if;

  -- Atomic counters: the conflict update takes a row lock, so concurrent callers are serialized per key.
  insert into public.registration_limits as l (key, window_start, attempts)
  values ('u:' || v_uid::text, date_trunc('hour', now()), 1)
  on conflict (key, window_start) do update set attempts = l.attempts + 1
  returning l.attempts into v_user_attempts;

  v_ip := nullif(trim(split_part(coalesce((nullif(current_setting('request.headers', true), '')::json ->> 'x-forwarded-for'), ''), ',', 1)), '');
  if v_ip is not null then
    insert into public.registration_limits as l (key, window_start, attempts)
    values ('ip:' || v_ip, date_trunc('hour', now()), 1)
    on conflict (key, window_start) do update set attempts = l.attempts + 1
    returning l.attempts into v_ip_attempts;
  end if;

  delete from public.registration_limits where window_start < now() - interval '24 hours';

  -- Return, never raise, so the counter increments commit.
  if v_user_attempts > 5 or v_ip_attempts > 20 then
    return jsonb_build_object('ok', false, 'reason', 'rate_limited');
  end if;

  v_phone := regexp_replace(coalesce(p_phone, ''), '[\s\-().]', '', 'g');
  if v_phone !~ '^\+[1-9][0-9]{7,14}$'
     or (v_phone ~ '^\+20' and v_phone !~ '^\+20(10|11|12|15)[0-9]{8}$') then
    return jsonb_build_object('ok', false, 'reason', 'invalid_number');
  end if;

  insert into public.accounts (user_id, phone_e164)
  values (v_uid, v_phone)
  on conflict (user_id) do update set phone_e164 = excluded.phone_e164
    where public.accounts.phone_verified = false;

  select a.phone_e164, a.phone_verified into v_stored, v_verified
  from public.accounts a where a.user_id = v_uid;

  if v_stored = v_phone then
    return jsonb_build_object('ok', true);
  end if;
  return jsonb_build_object('ok', false, 'reason', 'already_verified');
end;
$$;

revoke all on function public.register_phone(text) from public, anon;
grant execute on function public.register_phone(text) to authenticated;
```

```sql
-- TAW-7 cleanup policy (spec section 8). Review before running; run as the table owner.
-- Activity signal: latest auth.sessions.refreshed_at. A user with no session timestamp is never deleted.
-- Never deleted: verified accounts, users that are no longer anonymous, users active within the window.
delete from auth.users u
where u.is_anonymous
  and not exists (select 1 from public.accounts a where a.user_id = u.id and a.phone_verified)
  and (select max(s.refreshed_at) from auth.sessions s where s.user_id = u.id) is not null
  and (
    (not exists (select 1 from public.accounts a where a.user_id = u.id)
       and u.created_at < now() - interval '24 hours'
       and (select max(s.refreshed_at) from auth.sessions s where s.user_id = u.id) < now() - interval '24 hours')
    or
    (exists (select 1 from public.accounts a where a.user_id = u.id)
       and (select max(s.refreshed_at) from auth.sessions s where s.user_id = u.id) < now() - interval '30 days')
  );
```

```ts
export type Country = { iso: string; name: string; dial: string };

// Bundled static data; extending the list is data-only (spec section 5).
export const countries: readonly Country[] = [
  { iso: 'EG', name: 'Egypt', dial: '+20' },
  { iso: 'SA', name: 'Saudi Arabia', dial: '+966' },
  { iso: 'AE', name: 'United Arab Emirates', dial: '+971' },
];
export const defaultCountry: Country = countries[0];

export type PhoneResult = { ok: true; e164: string } | { ok: false };

export function normalizePhone(dial: string, national: string): PhoneResult {
  const digits = national.replace(/[\s\-().]/g, '');
  if (!/^[0-9]+$/.test(digits)) return { ok: false };
  const e164 = `${dial}${digits.replace(/^0/, '')}`;
  if (!/^\+[1-9][0-9]{7,14}$/.test(e164)) return { ok: false };
  if (dial === '+20' && !/^\+20(10|11|12|15)[0-9]{8}$/.test(e164)) return { ok: false };
  return { ok: true, e164 };
}
```

```ts
import * as SecureStore from 'expo-secure-store';

type Store = Pick<typeof SecureStore, 'getItemAsync' | 'setItemAsync' | 'deleteItemAsync'>;

// Values are split so that no single secure-store entry exceeds the historical 2048-byte limit.
export function createSecureStorage(store: Store, chunkSize = 1800) {
  const countKey = (k: string) => `${k}.n`;
  const partKey = (k: string, i: number) => `${k}.${i}`;

  async function removeItem(key: string) {
    const n = Number(await store.getItemAsync(countKey(key)));
    await store.deleteItemAsync(countKey(key));
    for (let i = 0; i < (Number.isFinite(n) ? n : 0); i++) await store.deleteItemAsync(partKey(key, i));
  }

  return {
    async getItem(key: string): Promise<string | null> {
      const raw = await store.getItemAsync(countKey(key));
      if (raw === null) return null;
      const n = Number(raw);
      let out = '';
      for (let i = 0; i < n; i++) {
        const part = await store.getItemAsync(partKey(key, i));
        if (part === null) return null;
        out += part;
      }
      return out;
    },
    async setItem(key: string, value: string): Promise<void> {
      if (!/^[\x00-\x7F]*$/.test(value)) throw new Error('secure storage: non-ASCII value');
      await removeItem(key);
      const n = Math.max(1, Math.ceil(value.length / chunkSize));
      for (let i = 0; i < n; i++) await store.setItemAsync(partKey(key, i), value.slice(i * chunkSize, (i + 1) * chunkSize));
      await store.setItemAsync(countKey(key), String(n)); // written last: a partial write is never read
    },
    removeItem,
  };
}

export const secureStorage = createSecureStorage(SecureStore);
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';

export type RegisterOutcome =
  | { status: 'registered' }
  | { status: 'invalid_number' | 'rate_limited' | 'already_verified' | 'offline' | 'failed' };

export type RegistrationState =
  | { kind: 'signed-out' }
  | { kind: 'needs-phone'; userId: string }
  | { kind: 'registered'; userId: string };

export type ConfirmResult = RegistrationState | { kind: 'unchanged' };

export type MarkerStore = { get(): Promise<string | null>; set(userId: string): Promise<void>; clear(): Promise<void> };
type Client = Pick<SupabaseClient, 'auth' | 'rpc' | 'from'>;

type ErrLike = { name?: string; message?: string; status?: number; code?: string } | null | undefined;

export function isNetworkError(e: ErrLike): boolean {
  return !!e && (e.name === 'AuthRetryableFetchError' || /network request failed|failed to fetch|timed out/i.test(e.message ?? ''));
}
function isAuthorizationError(e: ErrLike): boolean {
  return !!e && (e.status === 401 || e.code === 'PGRST301' || /jwt/i.test(e.message ?? ''));
}
function isTemporary(e: ErrLike): boolean {
  return isNetworkError(e) || (typeof e?.status === 'number' && e.status >= 500);
}

export async function registerPhone(client: Client, e164: string): Promise<RegisterOutcome> {
  const { data: current } = await client.auth.getSession();
  if (!current.session) {
    const { error } = await client.auth.signInAnonymously();
    if (error) return { status: isNetworkError(error) ? 'offline' : 'failed' };
  }
  const { data, error } = await client.rpc('register_phone', { p_phone: e164 });
  if (error) return { status: isNetworkError(error) ? 'offline' : 'failed' };
  if (data?.ok === true) return { status: 'registered' };
  const reason = data?.reason;
  if (reason === 'invalid_number' || reason === 'rate_limited' || reason === 'already_verified') return { status: reason };
  return { status: 'failed' };
}

export async function readInitialState(client: Client, marker: MarkerStore): Promise<RegistrationState> {
  const { data } = await client.auth.getSession();
  const session = data.session;
  if (!session) return { kind: 'signed-out' };
  const stored = await marker.get();
  return stored === session.user.id
    ? { kind: 'registered', userId: session.user.id }
    : { kind: 'needs-phone', userId: session.user.id };
}

async function readOwnRow(client: Client, userId: string) {
  return client.from('accounts').select('user_id').eq('user_id', userId).maybeSingle();
}

export async function confirmRegistration(client: Client, userId: string, marker: MarkerStore): Promise<ConfirmResult> {
  let { data, error } = await readOwnRow(client, userId);
  if (error && isAuthorizationError(error)) {
    const refreshed = await client.auth.refreshSession();
    if (refreshed.error) {
      if (isTemporary(refreshed.error)) return { kind: 'unchanged' };
      await client.auth.signOut({ scope: 'local' });
      await marker.clear();
      return { kind: 'signed-out' };
    }
    ({ data, error } = await readOwnRow(client, userId));
    if (error && isAuthorizationError(error)) {
      await client.auth.signOut({ scope: 'local' });
      await marker.clear();
      return { kind: 'signed-out' };
    }
  }
  if (error) return { kind: 'unchanged' };
  if (data) {
    await marker.set(userId);
    return { kind: 'registered', userId };
  }
  await marker.clear();
  return { kind: 'needs-phone', userId };
}
```
